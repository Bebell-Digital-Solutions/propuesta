                });

                if (!response.ok) throw new Error("Network response was not ok");
                
                const data = await response.json();
                
                // Assumes your backend returns JSON like: { "text": "## Markdown Result..." }
                // Adjust this depending on exactly how your Node.js server sends data back.
                return data.text || data.response || data.choices[0].message.content; 

            } catch (err) {
                console.warn("Could not connect to actual proxy. Using Fallback Simulation Mode for demonstration.");
